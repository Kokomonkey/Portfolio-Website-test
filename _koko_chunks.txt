/* ── HELODA KANBAN ─────────────────────────────────────────────────── */
const DEFAULT_VERSIONS = [
    { id: 'v01', name: 'v0.1 Core loop',    description: 'Plant, grow, harvest one plant.', order: 1 },
    { id: 'v02', name: 'v0.2 Warehouse',    description: 'Build & upgrade tiles.',           order: 2 },
    { id: 'v03', name: 'v0.3 Economy',      description: 'Buyers, prices, sales.',           order: 3 },
    { id: 'v04', name: 'v0.4 Variety',      description: 'Multiple plant types.',            order: 4 },
    { id: 'v05', name: 'v0.5 Risk',         description: 'Heat/police mechanic.',            order: 5 },
    { id: 'v06', name: 'v0.6 Progression',  description: 'Upgrades tree.',                   order: 6 },
    { id: 'v07', name: 'v0.7 Polish',       description: 'Sound, particles, tutorial.',      order: 7 },
];

function seedHeloda() {
    if (!state.heloda) state.heloda = { versions: [], cards: [] };
    if (!state.heloda.versions) state.heloda.versions = [];
    if (!state.heloda.cards) state.heloda.cards = [];
    if (!state.heloda.versions.length && !state.heloda._seeded) {
        state.heloda.versions = DEFAULT_VERSIONS.map(v => ({...v, updated_at: nowTs()}));
        state.heloda._seeded = true;
        saveState();
    }
}

function renderHeloda() {
    seedHeloda();
    const kanban = document.getElementById('kanban');
    if (!kanban) return;
    const versions = state.heloda.versions
        .filter(v => !v.deleted)
        .sort((a, b) => (a.order || 0) - (b.order || 0));
    kanban.innerHTML = versions.map(v => renderHelodaColumn(v)).join('');
}

function renderHelodaColumn(v) {
    const cards = (state.heloda.cards || [])
        .filter(c => !c.deleted && c.version_id === v.id)
        .sort((a, b) => (a.order || 0) - (b.order || 0));
    const done = cards.filter(c => c.status === 'done').length;
    const pct = cards.length ? Math.round(done / cards.length * 100) : 0;
    const cardsHtml = cards.map(c => renderHelodaCard(c)).join('') || '<div style="color:var(--text-muted); font-size:0.8em; text-align:center; padding:8px 0; font-style:italic;">Empty</div>';
    return '<div class="kanban-col" ondragover="cardDragOver(event)" ondragleave="cardDragLeave(event)" ondrop="cardDrop(event, \'' + v.id + '\')">'
        + '<div class="kanban-col-head">'
        + '<div class="name" onclick="editVersion(\'' + v.id + '\')" title="Click to rename">' + escapeHtml(v.name) + '</div>'
        + '<button class="icon-btn" onclick="deleteVersion(\'' + v.id + '\')" title="Delete version">&times;</button>'
        + '</div>'
        + '<div class="kanban-col-progress"><div class="fill" style="width:' + pct + '%"></div></div>'
        + '<div class="kanban-col-count">' + done + '/' + cards.length + ' &middot; ' + pct + '%</div>'
        + '<div class="kanban-cards">' + cardsHtml + '</div>'
        + '<button class="kanban-add-card" onclick="addCardTo(\'' + v.id + '\')">+ Add card</button>'
        + '</div>';
}

function renderHelodaCard(c) {
    const statusIcon  = { idea: '&#128161;', progress: '&#128295;', done: '&#9989;' }[c.status] || '&#128161;';
    const statusLabel = { idea: 'Idea', progress: 'In progress', done: 'Done' }[c.status] || 'Idea';
    const subs = c.subtasks || [];
    const subDone = subs.filter(s => s.done).length;
    return '<div class="kanban-card status-' + (c.status || 'idea') + '" draggable="true"'
        + ' ondragstart="cardDragStart(event, \'' + c.id + '\')" ondragend="cardDragEnd(event)"'
        + ' onclick="editCard(\'' + c.id + '\')">'
        + '<div class="card-title">' + escapeHtml(c.title) + '</div>'
        + '<div class="card-status">' + statusIcon + ' ' + statusLabel + '</div>'
        + (c.description ? '<div class="card-desc">' + escapeHtml(c.description) + '</div>' : '')
        + (subs.length ? '<div class="card-checklist">&#10003; ' + subDone + '/' + subs.length + ' sub-tasks</div>' : '')
        + '</div>';
}

/* Drag handlers */
function cardDragStart(e, id) {
    e.dataTransfer.setData('text/plain', id);
    e.dataTransfer.effectAllowed = 'move';
    e.currentTarget.classList.add('dragging');
    e.stopPropagation();
}
function cardDragEnd(e) {
    e.currentTarget.classList.remove('dragging');
}
function cardDragOver(e) {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    e.currentTarget.classList.add('drag-target');
}
function cardDragLeave(e) {
    if (e.currentTarget.contains(e.relatedTarget)) return;
    e.currentTarget.classList.remove('drag-target');
}
function cardDrop(e, versionId) {
    e.preventDefault();
    e.currentTarget.classList.remove('drag-target');
    const id = e.dataTransfer.getData('text/plain');
    const card = state.heloda.cards.find(c => c.id === id);
    if (card && card.version_id !== versionId) {
        card.version_id = versionId;
        card.updated_at = nowTs();
        saveState();
        renderHeloda();
    }
}

function addCardTo(versionId) {
    const title = prompt('Card title:');
    if (!title || !title.trim()) return;
    state.heloda.cards.push({
        id: uid(),
        title: title.trim(),
        description: '',
        status: 'idea',
        version_id: versionId,
        subtasks: [],
        order: (state.heloda.cards.length + 1) * 10,
        updated_at: nowTs(),
    });
    saveState();
    renderHeloda();
}

/* Card edit modal */
let editingCardId = null;

function editCard(id) {
    const c = state.heloda.cards.find(x => x.id === id);
    if (!c) return;
    editingCardId = id;
    document.getElementById('card-title').value = c.title || '';
    document.getElementById('card-desc').value = c.description || '';
    document.getElementById('card-status').value = c.status || 'idea';
    renderSubtasks();
    document.getElementById('cardModal').classList.add('open');
}
function closeCardModal() {
    document.getElementById('cardModal').classList.remove('open');
    editingCardId = null;
}
function renderSubtasks() {
    const c = state.heloda.cards.find(x => x.id === editingCardId);
    if (!c) return;
    const list = document.getElementById('card-subtasks');
    const subs = c.subtasks || [];
    if (!subs.length) {
        list.innerHTML = '<div style="color:var(--text-muted); font-size:0.85em; font-style:italic; padding:4px 0;">No sub-tasks yet.</div>';
        return;
    }
    list.innerHTML = subs.map((s, i) =>
        '<div style="display:flex; align-items:center; gap:6px; margin-bottom:4px;">'
        + '<input type="checkbox" ' + (s.done ? 'checked' : '') + ' onchange="toggleSubtask(' + i + ', this.checked)">'
        + '<input type="text" value="' + escapeHtml(s.text) + '"'
        + ' oninput="updateSubtaskText(' + i + ', this.value)"'
        + ' style="flex:1; padding:4px 8px; border:1px solid var(--border); border-radius:4px; background:var(--bg); color:var(--text); font-family:inherit;">'
        + '<button onclick="removeSubtask(' + i + ')" style="background:none; border:none; color:var(--red); cursor:pointer; font-family:inherit; font-size:1.1em; padding:0 4px;">&times;</button>'
        + '</div>'
    ).join('');
}
function addSubtask() {
    const c = state.heloda.cards.find(x => x.id === editingCardId);
    if (!c) return;
    c.subtasks = c.subtasks || [];

];
const IDEA_COLORS = ['yellow', 'pink', 'blue', 'green', 'orange', 'purple'];

function hashCode(s) {
    let h = 0;
    for (let i = 0; i < s.length; i++) h = ((h << 5) - h) + s.charCodeAt(i) | 0;
    return Math.abs(h);
}

function renderBook() {
    if (!state.book) state.book = { quests: [], ideas: [] };
    if (!state.book.quests) state.book.quests = [];
    if (!state.book.ideas) state.book.ideas = [];
    renderBookProgress();
    renderQuests();
    renderIdeas();
}

function renderBookProgress() {
    const el = document.getElementById('book-progress');
    if (!el) return;
    const answered = new Set();
    for (const q of (state.book.quests || [])) {
        if (!q.deleted && q.answer && q.answer.trim()) answered.add(q.prompt_key);
    }
    const total = BOOK_QUESTS.length;
    const done = answered.size;
    el.innerHTML = '<div class="num">' + done + '/' + total + '</div>'
        + '<div><div style="font-weight:700; color:var(--primary-dark);">quests answered</div>'
        + '<div class="lbl">' + Math.round(done / total * 100) + '% of your idea foundation</div></div>';
}

function renderQuests() {
    const list = document.getElementById('quest-list');
    if (!list) return;
    const answers = {};
    for (const q of (state.book.quests || [])) {
        if (q.deleted) continue;
        if (!answers[q.prompt_key] || q.updated_at > answers[q.prompt_key].updated_at) {
            answers[q.prompt_key] = q;
        }
    }
    list.innerHTML = BOOK_QUESTS.map((q, i) => {
        const existing = answers[q.key];
        const answer = existing ? (existing.answer || '') : '';
        const answered = !!answer.trim();
        return '<div class="quest-card ' + (answered ? 'answered' : '') + '">'
            + '<div style="display:flex; justify-content:space-between; align-items:baseline;">'
            + '<div class="quest-num">Quest ' + (i + 1) + ' / ' + BOOK_QUESTS.length + '</div>'
            + (answered ? '<span class="answered-badge">&#10003; Answered</span>' : '')
            + '</div>'
            + '<div class="quest-prompt">' + escapeHtml(q.prompt) + '</div>'
            + '<textarea id="quest-input-' + q.key + '" placeholder="Write your answer…">' + escapeHtml(answer) + '</textarea>'
            + '<div class="actions">'
            + '<button class="btn btn-primary btn-sm" onclick="saveQuest(\'' + q.key + '\')">Save</button>'
            + (answered ? '<button class="btn btn-ghost btn-sm" onclick="clearQuest(\'' + q.key + '\')">Clear</button>' : '')
            + '</div>'
            + '</div>';
    }).join('');
}

function saveQuest(key) {
    const input = document.getElementById('quest-input-' + key);
    if (!input) return;
    const text = input.value.trim();
    for (const q of (state.book.quests || [])) {
        if (!q.deleted && q.prompt_key === key) {
            q.deleted = true;
            q.updated_at = nowTs();
        }
    }
    if (text) {
        state.book.quests.push({
            id: uid(),
            prompt_key: key,
            answer: text,
            updated_at: nowTs(),
        });
    }
    saveState();
    renderBookProgress();
    renderQuests();
}
function clearQuest(key) {
    if (!confirm('Clear your answer for this quest?')) return;
    for (const q of (state.book.quests || [])) {
        if (!q.deleted && q.prompt_key === key) {
            q.deleted = true;
            q.updated_at = nowTs();
        }
    }
    saveState();
    renderBookProgress();
    renderQuests();
}

function renderIdeas() {
    const board = document.getElementById('idea-board');
    if (!board) return;
    const items = (state.book.ideas || [])
        .filter(i => !i.deleted)
        .sort((a, b) => (b.updated_at || 0) - (a.updated_at || 0));
    if (!items.length) {
        board.innerHTML = '<div style="grid-column:1/-1; text-align:center; color:#fff5db; padding:40px 20px; font-family:Georgia, serif; font-style:italic;">The board is empty. Pin an idea above.</div>';
        return;
    }
    board.innerHTML = items.map(idea => {
        const h = hashCode(idea.id);
        const color = IDEA_COLORS[h % IDEA_COLORS.length];
        const rot = ((h % 7) - 3);
        return '<div class="idea-note ' + color + '" style="--rot:' + rot + 'deg;">'
            + '<button class="idea-del" onclick="deleteIdea(\'' + idea.id + '\')" title="Remove">&times;</button>'
            + '<div class="idea-text">' + escapeHtml(idea.text) + '</div>'
            + (idea.tag ? '<div class="idea-tag">' + escapeHtml(idea.tag) + '</div>' : '')
            + '</div>';
    }).join('');
}

function addIdea(e) {
    e.preventDefault();
    const f = e.target;
    if (!state.book.ideas) state.book.ideas = [];
    state.book.ideas.push({
        id: uid(),
        text: f.text.value.trim(),
        tag: f.tag.value.trim(),
        updated_at: nowTs(),
    });
    saveState();
    f.reset();
    renderIdeas();
}
function deleteIdea(id) {
    const i = state.book.ideas.find(x => x.id === id);
    if (i) {
        i.deleted = true;
        i.updated_at = nowTs();
        saveState();
        renderIdeas();
    }
}
