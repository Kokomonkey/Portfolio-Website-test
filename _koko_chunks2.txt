/* Helpers */
function getNodeSubs(node, nodeState) {
    const stored = nodeState[node.id];
    if (stored && Array.isArray(stored.subtasks)) return stored.subtasks;
    return (node.subtasks || []).map(s => ({...s}));
}

function computeNodeStatus(node, spec, nodeState) {
    const subs = getNodeSubs(node, nodeState);
    let progress = 0;
    if (subs.length) {
        const done = subs.filter(s => s.done).length;
        progress = (done / subs.length) * 100;
    } else if ((nodeState[node.id] || {}).completed) {
        progress = 100;
    }
    const prereqs = node.prereqs || [];
    let prereqsDone = true;
    for (const pid of prereqs) {
        const pn = spec.find(x => x.id === pid);
        if (!pn) continue;
        const psubs = getNodeSubs(pn, nodeState);
        if (psubs.length) {
            if (!psubs.every(s => s.done)) { prereqsDone = false; break; }
        } else if (!(nodeState[pn.id] || {}).completed) {
            prereqsDone = false;
            break;
        }
    }
    let status;
    if (progress >= 100) status = 'completed';
    else if (progress > 0) status = 'in-progress';
    else if (prereqsDone) status = 'available';
    else status = 'locked';
    return { status: status, progress: progress };
}

function renderTree(kind) {
    const spec = kind === 'portfolio' ? PORTFOLIO_TREE : MANGA_TREE;
    const edges = kind === 'portfolio' ? PORTFOLIO_EDGES : MANGA_EDGES;
    const svg = document.getElementById(kind + '-tree-svg');
    if (!svg) return;
    if (!state[kind]) state[kind] = {};
    if (!state[kind].nodeState) state[kind].nodeState = {};
    const nodeState = state[kind].nodeState;

    const statuses = {};
    for (const n of spec) statuses[n.id] = computeNodeStatus(n, spec, nodeState);

    let edgesHtml = '<g class="edges">';
    for (const e of edges) {
        const from = spec.find(n => n.id === e.from);
        const to = spec.find(n => n.id === e.to);
        if (!from || !to) continue;
        const targetStatus = statuses[to.id].status;
        const cls = targetStatus === 'locked' ? 'edge-dashed' : 'edge-solid';
        const fromSize = from.size || 24;
        const toSize = to.size || 24;
        const fy = from.y + fromSize;
        const ty = to.y - toSize;
        const ey = Math.round((fy + ty) / 2);
        let d;
        if (from.x === to.x) {
            d = 'M ' + from.x + ' ' + fy + ' L ' + to.x + ' ' + ty;
        } else {
            d = 'M ' + from.x + ' ' + fy + ' L ' + from.x + ' ' + ey + ' L ' + to.x + ' ' + ey + ' L ' + to.x + ' ' + ty;
        }
        edgesHtml += '<path class="' + cls + '" d="' + d + '"/>';
    }
    edgesHtml += '</g>';

    let nodesHtml = '';
    for (const n of spec) nodesHtml += renderTreeNode(kind, n, statuses[n.id], nodeState);
    svg.innerHTML = edgesHtml + nodesHtml;
}

function renderTreeNode(kind, node, status, nodeState) {
    const r = node.size || 24;
    const x = node.x;
    const y = node.y;
    const labelY = y + r + 20;
    const stored = nodeState[node.id] || {};
    const isFocus = !!stored.is_focus;
    let inner = '';
    if (status.status === 'completed') {
        inner = '<circle cx="' + x + '" cy="' + y + '" r="' + r + '" fill="#e6b85a" stroke="#a67a2a" stroke-width="3"/>'
              + '<text class="node-icon" text-anchor="middle" x="' + x + '" y="' + (y + 6) + '">&#10003;</text>';
    } else if (status.status === 'in-progress') {
        let ringHtml = '';
        const pct = status.progress / 100;
        if (pct > 0.001 && pct < 0.999) {
            const angle = pct * 2 * Math.PI;
            const endX = x + r * Math.sin(angle);
            const endY = y - r * Math.cos(angle);
            const largeArc = pct > 0.5 ? 1 : 0;
            ringHtml = '<path d="M ' + x + ' ' + y + ' L ' + x + ' ' + (y - r)
                     + ' A ' + r + ' ' + r + ' 0 ' + largeArc + ' 1 ' + endX.toFixed(2) + ' ' + endY.toFixed(2)
                     + ' Z" fill="#d4a373" opacity="0.55"/>';
        }
        inner = '<circle cx="' + x + '" cy="' + y + '" r="' + r + '" fill="#faf4ec" stroke="#d4a373" stroke-width="3"/>'
              + ringHtml
              + '<text class="node-pct" text-anchor="middle" x="' + x + '" y="' + (y + 4) + '">' + Math.round(status.progress) + '%</text>';
    } else if (status.status === 'available') {
        const icon = node.icon || '○';
        inner = '<circle cx="' + x + '" cy="' + y + '" r="' + r + '" fill="#faf4ec" stroke="#d4a373" stroke-width="2"/>'
              + '<text class="node-icon" text-anchor="middle" x="' + x + '" y="' + (y + 6) + '">' + icon + '</text>';
    } else {
        inner = '<circle cx="' + x + '" cy="' + y + '" r="' + r + '" fill="#e8dcc9" stroke="#a5967f" stroke-width="2" stroke-dasharray="3 2"/>'
              + '<text class="node-icon" text-anchor="middle" x="' + x + '" y="' + (y + 6) + '" opacity="0.5">&#128274;</text>';
    }
    const focusMark = isFocus ? '<circle cx="' + (x + r - 3) + '" cy="' + (y - r + 3) + '" r="5" fill="#e6b85a" stroke="#3d2f25" stroke-width="1.5"/>' : '';
    return '<g class="tree-node ' + status.status + '" onclick="openNodeModal(\'' + kind + '\', \'' + node.id + '\')">'
        + inner
        + focusMark
        + '<text class="node-label" text-anchor="middle" x="' + x + '" y="' + labelY + '">' + escapeHtml(node.label) + '</text>'
        + '</g>';
}

let currentNodeCtx = null;
let modalSubs = [];

function openNodeModal(kind, id) {
    const spec = kind === 'portfolio' ? PORTFOLIO_TREE : MANGA_TREE;
    const node = spec.find(n => n.id === id);
    if (!node) return;
    currentNodeCtx = { kind: kind, id: id };
    if (!state[kind]) state[kind] = {};
    if (!state[kind].nodeState) state[kind].nodeState = {};
    const nodeState = state[kind].nodeState;
    const stored = nodeState[id] || {};
    modalSubs = (Array.isArray(stored.subtasks) ? stored.subtasks : (node.subtasks || []).map(s => ({...s}))).map(s => ({...s}));

    document.getElementById('nm-icon').innerHTML = node.icon || '&#8226;';
    document.getElementById('nm-title').textContent = node.label;
    document.getElementById('nm-desc').innerHTML = node.desc || '';
    document.getElementById('nm-notes').value = stored.notes || '';

    const prereqs = node.prereqs || [];
    let prereqsHtml = '';
    if (prereqs.length) {
        prereqsHtml = '<div class="prereq-list"><strong>Prerequisites:</strong> '
            + prereqs.map(pid => {
                const p = spec.find(n => n.id === pid);
                if (!p) return '';
                const ps = computeNodeStatus(p, spec, nodeState);
                let icon;
                if (ps.status === 'completed') icon = '&#10003;';
                else if (ps.status === 'in-progress') icon = Math.round(ps.progress) + '%';
                else if (ps.status === 'available') icon = '&#9711;';
                else icon = '&#128274;';
                return '<span class="prereq-chip prereq-' + ps.status + '">' + icon + ' ' + escapeHtml(p.label) + '</span>';
            }).join('')
            + '</div>';
    }
    document.getElementById('nm-prereqs').innerHTML = prereqsHtml;

    const isManga = kind === 'manga';
    document.getElementById('nm-focus-row').style.display = isManga ? 'block' : 'none';
    document.getElementById('nm-focus').checked = !!stored.is_focus;

    const status = computeNodeStatus(node, spec, nodeState);
    document.getElementById('nm-status').textContent = status.status.replace('-', ' ');
    document.getElementById('nm-progress-fill').style.width = Math.round(status.progress) + '%';
    document.getElementById('nm-progress-label').textContent = Math.round(status.progress) + '% complete';

    renderModalSubtasks();

    if (!state.counters) state.counters = {};
    state.counters.node_opens = (state.counters.node_opens || 0) + 1;
    saveState();

    document.getElementById('nodeModal').classList.add('open');
}

function renderModalSubtasks() {
    const list = document.getElementById('nm-subtasks');
    if (!modalSubs.length) {
        list.innerHTML = '<div style="color:var(--text-muted); font-size:0.85em; font-style:italic; padding:6px 0;">No sub-tasks yet. Add some to track progress.</div>';
        return;
    }
    list.innerHTML = modalSubs.map((s, i) =>
        '<div class="node-sub-item">'
        + '<input type="checkbox" ' + (s.done ? 'checked' : '') + ' onchange="toggleModalSub(' + i + ', this.checked)">'
        + '<input type="text" class="' + (s.done ? 'done' : '') + '" value="' + escapeHtml(s.text) + '" oninput="updateModalSubText(' + i + ', this.value)">'
        + '<button onclick="removeModalSub(' + i + ')" title="Remove">&times;</button>'
        + '</div>'
    ).join('');
}
function toggleModalSub(i, checked) { modalSubs[i].done = checked; renderModalSubtasks(); }
function updateModalSubText(i, text) { modalSubs[i].text = text; }
function removeModalSub(i) { modalSubs.splice(i, 1); renderModalSubtasks(); }
function addNodeSubtask() {

    renderModalSubtasks();
    setTimeout(() => {
        const inputs = document.querySelectorAll('#nm-subtasks input[type=text]');
        if (inputs.length) inputs[inputs.length - 1].focus();
    }, 30);
}

    if (!currentNodeCtx) return;
    const kind = currentNodeCtx.kind;

    if (!state[kind]) state[kind] = {};
    if (!state[kind].nodeState) state[kind].nodeState = {};

}

function getFocusSkills() {
    const focus = new Set();
    const ns = (state.manga && state.manga.nodeState) || {};
    for (const id in ns) {
        if (ns[id].is_focus) focus.add(id);
    }
    return focus;
}

function pickDailyPrompt() {
    const focus = getFocusSkills();
    const pool = [];
    for (let i = 0; i < MANGA_PROMPTS.length; i++) {
        pool.push(i);
        if (MANGA_PROMPTS[i].skill && focus.has(MANGA_PROMPTS[i].skill)) pool.push(i);
    }
    const todayIso = fmtISODate(new Date());
    const overrides = (state.manga && state.manga.promptOverrides) || {};
    if (overrides[todayIso] !== undefined && MANGA_PROMPTS[overrides[todayIso]]) {
        return { index: overrides[todayIso], prompt: MANGA_PROMPTS[overrides[todayIso]] };
    }
    const h = hashCode(todayIso);
    const idx = pool[h % pool.length];
    return { index: idx, prompt: MANGA_PROMPTS[idx] };
}

function renderDailyPrompt() {
    const el = document.getElementById('dp-text');
    if (!el) return;
    const picked = pickDailyPrompt();
    el.innerHTML = picked.prompt.text;
    const meta = document.getElementById('dp-meta');
    let skillName = 'General';
    if (picked.prompt.skill) {
        const n = MANGA_TREE.find(x => x.id === picked.prompt.skill);
        if (n) skillName = n.label;
    }
    const focus = getFocusSkills();
    const isFocus = picked.prompt.skill && focus.has(picked.prompt.skill);
    meta.innerHTML = '<span>Skill: <strong>' + escapeHtml(skillName) + '</strong></span>'
        + (isFocus ? '<span class="focus-badge">&#11088; Focus</span>' : '');
}

function rerollPrompt() {
    if (!state.manga) state.manga = {};
    if (!state.manga.promptOverrides) state.manga.promptOverrides = {};
    const todayIso = fmtISODate(new Date());
    let newIdx = Math.floor(Math.random() * MANGA_PROMPTS.length);
    const cur = pickDailyPrompt().index;
    if (cur === newIdx) newIdx = (newIdx + 1) % MANGA_PROMPTS.length;
    state.manga.promptOverrides[todayIso] = newIdx;
    saveState();
    renderDailyPrompt();
}

function saveSession() {
    const picked = pickDailyPrompt();
    if (!state.manga) state.manga = {};
    if (!state.manga.promptLog) state.manga.promptLog = [];
    state.manga.promptLog.push({
        id: uid(),
        date: fmtISODate(new Date()),
        prompt_text: picked.prompt.text,
        skill: picked.prompt.skill || null,
        rating: sessionRating || null,
        notes: document.getElementById('session-notes').value.trim(),
        updated_at: nowTs(),
    });
    saveState();
    closeSessionModal();
    renderSessionLog();
}

function renderSessionLog() {
    const el = document.getElementById('session-log');
    if (!el) return;
    const log = ((state.manga && state.manga.promptLog) || [])
        .filter(l => !l.deleted)
        .sort((a, b) => (b.updated_at || 0) - (a.updated_at || 0))
        .slice(0, 10);
    if (!log.length) {
        el.innerHTML = '<div style="color:var(--text-muted); font-size:0.85em; font-style:italic; padding: 10px 0;">No sessions logged yet. Log your first practice above.</div>';
        return;
    }
    el.innerHTML = log.map(l => {
        const rating = l.rating || 0;
        const stars = '★'.repeat(rating) + '☆'.repeat(5 - rating);
        return '<div class="session-item">'
            + '<div class="session-date">' + relativeTime(l.updated_at || 0) + '</div>'
            + '<div class="session-rating">' + stars + '</div>'
            + '<div class="session-prompt">' + l.prompt_text + '</div>'
            + (l.notes ? '<div class="session-notes">' + escapeHtml(l.notes) + '</div>' : '')
            + '</div>';
    }).join('');
}

function relativeTime(ts) {
    if (!ts) return '';
    const diff = Date.now() - ts;
    if (diff < 60000) return 'just now';
    if (diff < 3600000) return Math.floor(diff/60000) + 'm ago';
    if (diff < 86400000) return Math.floor(diff/3600000) + 'h ago';
    return Math.floor(diff/86400000) + 'd ago';
}
