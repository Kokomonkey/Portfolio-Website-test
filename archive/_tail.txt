function updateSyncStatusUI() {
    const el = document.getElementById('sync-status');
    if (!el) return;
    if (!syncConfigured()) {
        el.textContent = '⚙️ Setup sync';
        el.className = 'sync-status not-setup';
        return;
    }
    const labels = {
        idle:    '☁️ Synced',
        syncing: '⟳ Syncing…',
        error:   '⚠️ Sync error — tap to fix',
    };
    el.textContent = labels[syncStatus] || labels.idle;
    el.className = 'sync-status ' + syncStatus;
}

function setSyncStatus(s) {
    syncStatus = s;
    updateSyncStatusUI();
}

function mergeStates(a, b) {
    const lists = ['tasks', 'books', 'movies', 'series', 'anime', 'songs'];
    const out = {};
    for (const k of lists) {
        const map = new Map();
        for (const it of (a[k] || [])) map.set(it.id, it);
        for (const it of (b[k] || [])) {
            const ex = map.get(it.id);
            if (!ex) { map.set(it.id, it); continue; }
            if (it.deleted && !ex.deleted) { map.set(it.id, it); continue; }
            if (ex.deleted && !it.deleted) { continue; }
            const eU = ex.updated_at || 0;
            const iU = it.updated_at || 0;
            if (iU > eU) map.set(it.id, it);
        }
        out[k] = Array.from(map.values());
    }
    return out;
}

function rerenderActive() {
    const active = document.querySelector('.page.active');
    if (!active) return;
    const name = active.id.replace('page-', '');
    if (name === 'calendar') renderCalendar();
    if (name === 'books') renderBooks();
    if (name === 'movies') renderMovies();
    if (name === 'series') renderSeries();
    if (name === 'anime') renderAnime();
    if (name === 'songs') renderSongs();
}

async function fetchRemote() {
    const r = await fetch(JSONBIN_BASE + '/' + syncBin + '/latest', {
        headers: { 'X-Master-Key': syncKey, 'X-Bin-Meta': 'false' },
        cache: 'no-store',
    });
    if (!r.ok) throw new Error('GET HTTP ' + r.status);
    const remote = await r.json();
    return (remote && typeof remote === 'object') ? remote : {};
}

async function pullSync() {
    if (!syncConfigured()) return;
    setSyncStatus('syncing');
    try {
        const remote = await fetchRemote();
        state = mergeStates(state, remote);
        localStorage.setItem(STORE_KEY, JSON.stringify(state));
        rerenderActive();
        setSyncStatus('idle');
    } catch (e) {
        console.warn('pullSync failed:', e);
        setSyncStatus('error');
    }
}

function scheduleSyncPush() {
    if (!syncConfigured()) return;
    if (pushTimer) clearTimeout(pushTimer);
    pushTimer = setTimeout(pushSync, 700);
}

async function pushSync() {
    if (!syncConfigured() || pushInFlight) return;
    pushInFlight = true;
    setSyncStatus('syncing');
    try {
        let remote = {};
        try { remote = await fetchRemote(); } catch (_) {}
        const merged = mergeStates(remote, state);
        const r = await fetch(JSONBIN_BASE + '/' + syncBin, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json', 'X-Master-Key': syncKey },
            body: JSON.stringify(merged),
        });
        if (!r.ok) throw new Error('PUT HTTP ' + r.status);
        state = merged;
        localStorage.setItem(STORE_KEY, JSON.stringify(state));
        rerenderActive();
        setSyncStatus('idle');
    } catch (e) {
        console.warn('pushSync failed:', e);
        setSyncStatus('error');
    } finally {
        pushInFlight = false;
    }
}

function startPullPolling() {
    if (pullTimer) clearInterval(pullTimer);
    if (!syncConfigured()) return;
    pullTimer = setInterval(() => {
        if (document.visibilityState === 'visible') pullSync();
    }, 15000);
}

function openSyncSetup() {
    document.getElementById('sync-bin-input').value = syncBin;
    document.getElementById('sync-key-input').value = syncKey;
    document.getElementById('syncModal').classList.add('open');
}
function closeSyncSetup() {
    document.getElementById('syncModal').classList.remove('open');
}
function saveSyncConfig() {
    const bin = document.getElementById('sync-bin-input').value.trim();
    const key = document.getElementById('sync-key-input').value.trim();
    if (!bin || !key) { alert('Both Bin ID and Master Key are required.'); return; }
    syncBin = bin;
    syncKey = key;
    localStorage.setItem(SYNC_BIN_STORAGE, syncBin);
    localStorage.setItem(SYNC_KEY_STORAGE, syncKey);
    closeSyncSetup();
    updateSyncStatusUI();
    startPullPolling();
    pushSync();
}
function clearSyncConfig() {
    if (!confirm('Disconnect sync? Your local data stays on this device.')) return;
    syncBin = '';
    syncKey = '';
    localStorage.removeItem(SYNC_BIN_STORAGE);
    localStorage.removeItem(SYNC_KEY_STORAGE);
    if (pullTimer) clearInterval(pullTimer);
    setSyncStatus('idle');
    updateSyncStatusUI();
    closeSyncSetup();
}

document.addEventListener('visibilitychange', () => {
    if (document.visibilityState === 'visible' && syncConfigured()) pullSync();
});

let lastPageShown = null;
function showPageIntro(name) {
    if (name === lastPageShown) return;
    lastPageShown = name;
    const svg = PAGE_INTROS[name];
    if (!svg) return;
    const intro = document.createElement('div');
    intro.className = 'page-intro intro-' + name;
    intro.innerHTML = svg;
    document.body.appendChild(intro);
    setTimeout(() => intro.remove(), 1950);
}

function showWelcome(person) {
    const labels = { douwe: 'Douwe', ilian: 'Ilian' };
    if (!labels[person]) return;
    const overlay = document.createElement('div');
    overlay.className = 'welcome-overlay ' + person;
    overlay.innerHTML = '<div class="welcome-text">Welcome ' + labels[person] + '</div>';
    document.body.appendChild(overlay);
    setTimeout(() => overlay.remove(), 2500);
}
