function renderBooks() {
    const grid = document.getElementById('books-grid');
    const items = state.books.filter(b => !b.deleted);
    const toolbar = document.getElementById('books-toolbar');
    if (toolbar) {
        toolbar.innerHTML = '<div class="view-toggle">'
            + '<button class="' + (booksView==='grid'?'active':'') + '" onclick="setBooksView(\'grid\')">Grid</button>'
            + '<button class="' + (booksView==='bookcase'?'active':'') + '" onclick="setBooksView(\'bookcase\')">&#128218; Bookcase</button>'
            + '</div>';
    }
    if (booksView === 'bookcase') {
        grid.style.display = 'block';
        grid.innerHTML = renderBookcase(items);
        return;
    }
    grid.style.display = '';
    if (!items.length) {
        grid.innerHTML = '<div class="empty-state" style="grid-column: 1/-1;"><div class="empty-icon">&#128214;</div>Add your first book above.</div>';
        return;
    }
    let html = '';
    for (const b of items) {
        html += '<div class="lib-card">';
        html += coverHtml(b.cover_url, '&#128214;');
        html += '<div class="nm">' + escapeHtml(b.title) + '</div>';
        html += '<div class="meta">' + escapeHtml(b.author || '-') + '</div>';
        html += '<progress value="' + b.pages_read + '" max="' + b.total_pages + '"></progress>';
        html += '<div class="meta">' + b.pages_read + '/' + b.total_pages + ' pages</div>';
        html += '<div class="tag-row">' + personTag(b.person) + '</div>';
        html += '<div class="actions">';
        html += '<button class="btn btn-ghost btn-sm" onclick="updateBookProgress(\'' + b.id + '\')">Update</button>';
        html += '<button class="btn btn-link btn-sm" onclick="deleteItem(\'books\',\'' + b.id + '\')">&times;</button>';
        html += '</div></div>';
    }
    grid.innerHTML = html;
}

const COUNTDOWN_TARGET = new Date('2026-09-01T00:00:00');
function renderCountdown() {
    const el = document.getElementById('countdown-widget');
    if (!el) return;

const BOOKCASE_PALETTE = ['#8b3a3a','#6b4226','#3d5a47','#5c3a5e','#4a5d7a','#7d6b3a','#6b2e2e','#3a5a5a','#8b5a3c','#9b6b3c','#7a5a8a','#5a7a8a'];
function renderBookcase(items) {
    if (!items.length) {
        return '<div class="bookcase"><div class="empty-state" style="color:#e6c89a;"><div class="empty-icon">&#128218;</div>No books on the shelf yet.</div></div>';
    }
    const finishedCount = items.filter(b => b.is_finished).length;
    let html = '<div class="bookcase">';
    html += '<div class="bookcase-stats"><span class="title">Our library</span>'
        + '<span>' + finishedCount + ' / ' + items.length + ' read &middot; '
        + (items.length ? Math.round(finishedCount / items.length * 100) : 0) + '%</span></div>';
    html += '<div class="shelf">';
    items.forEach((b, i) => {
        const pages = b.total_pages || 200;
        const width = Math.min(42, 16 + Math.floor(pages / 50));
        const height = Math.min(140, 95 + Math.floor(pages / 12));
        const color = BOOKCASE_PALETTE[i % BOOKCASE_PALETTE.length];
        const klass = b.is_finished ? 'finished' : (b.pages_read === 0 ? 'unread' : '');
        html += '<div class="book-spine ' + klass + '" '
            + 'style="width:' + width + 'px;height:' + height + 'px;'
            + '--spine-color:' + color + ';--delay:' + (i*0.04) + 's"'
            + ' onclick="updateBookProgress(\'' + b.id + '\')"'
            + ' title="' + escapeAttr(b.title) + ' (' + b.pages_read + '/' + b.total_pages + ')">'
            + '<span class="spine-title">' + escapeHtml(b.title) + '</span></div>';
    });
    html += '</div></div>';
    return html;
}

// === WEEKLY ===
function getMondayOf(d) {
    const date = new Date(d);
    date.setHours(0,0,0,0);
    const day = date.getDay();
    const diff = (day === 0 ? -6 : 1 - day);
    date.setDate(date.getDate() + diff);
    return date;
}
function getWeekStartISO(d) { return fmtISODate(getMondayOf(d || new Date())); }

function renderWeekly() {
    if (!state.weeklies) state.weeklies = [];
    const today = new Date();
    const thisWeek = getWeekStartISO(today);

    // Find current topic & answers
    const items = state.weeklies.filter(w => !w.deleted);
    const topicItems = items.filter(w => w.kind === 'topic' && w.week_start === thisWeek)
                            .sort((a,b) => (b.updated_at||0) - (a.updated_at||0));
    const currentTopic = topicItems[0] || null;
    const findAnswer = (person) => items
        .filter(w => w.kind === 'answer' && w.week_start === thisWeek && w.person === person)
        .sort((a,b) => (b.updated_at||0) - (a.updated_at||0))[0] || null;

    const douweAns = findAnswer('douwe');
    const ilianAns = findAnswer('ilian');

    const hero = document.getElementById('weekly-hero');
    const weekLabel = getMondayOf(today).toLocaleDateString('en-US', { month: 'long', day: 'numeric' })
        + ' — week ' + getISOWeek(today);

    let topicHtml;
    if (currentTopic) {
        topicHtml = '<div class="weekly-topic-text" onclick="editWeeklyTopic()" id="weekly-topic-display">'
            + escapeHtml(currentTopic.text) + '</div>'
            + '<div class="weekly-topic-meta">Set by ' + (currentTopic.person === 'douwe' ? 'Douwe' : 'Ilian') + ' &middot; click to edit</div>';
    } else {
        topicHtml = '<div class="weekly-topic-text empty" onclick="editWeeklyTopic()" id="weekly-topic-display">'
            + 'No topic yet — click to set one for this week.</div>';
    }
    topicHtml += '<div class="weekly-topic-editor" id="weekly-topic-editor">'
        + '<input type="text" id="weekly-topic-input" placeholder="What\'s this week\'s topic?">'
        + '<button class="btn btn-primary btn-sm" onclick="saveWeeklyTopic()">Save</button>'
        + '<button class="btn btn-ghost btn-sm" onclick="cancelWeeklyTopic()">Cancel</button>'
        + '</div>';

    const answerHtml = (person, ans) => {
        const isMe = person === currentPerson;
        const label = person === 'douwe' ? 'Douwe' : 'Ilian';
        let inner;
        if (isMe) {
            inner = '<textarea id="weekly-ans-' + person + '" placeholder="Write your answer...">'
                + (ans ? escapeHtml(ans.text) : '') + '</textarea>'
                + '<div style="margin-top:8px; text-align:right;">'
                + '<button class="btn btn-primary btn-sm" onclick="saveWeeklyAnswer(\'' + person + '\')">Save</button>'
                + '</div>';
        } else if (ans) {
            inner = '<div class="text">' + escapeHtml(ans.text) + '</div>';
        } else {
            inner = '<div class="empty-text">Not answered yet.</div>';
        }
        return '<div class="weekly-answer ' + person + '">'
            + '<div class="who"><span class="who-name">' + label + '</span>' + personTag(person) + '</div>'
            + inner + '</div>';
    };

    hero.innerHTML = '<div class="weekly-label">Week of ' + weekLabel + '</div>'
        + topicHtml
        + '<div class="weekly-answers">' + answerHtml('douwe', douweAns) + answerHtml('ilian', ilianAns) + '</div>';

    // Archive: all past weeks (not thisWeek), grouped by week_start desc
    const past = items.filter(w => w.week_start !== thisWeek);
    const byWeek = {};
    for (const w of past) {
        if (!byWeek[w.week_start]) byWeek[w.week_start] = { topic: null, answers: {} };
        if (w.kind === 'topic') {
            if (!byWeek[w.week_start].topic || w.updated_at > (byWeek[w.week_start].topic.updated_at || 0)) {
                byWeek[w.week_start].topic = w;
            }
        } else if (w.kind === 'answer') {
            const existing = byWeek[w.week_start].answers[w.person];
            if (!existing || w.updated_at > (existing.updated_at || 0)) {
                byWeek[w.week_start].answers[w.person] = w;
            }
        }
    }
    const weeks = Object.keys(byWeek).sort().reverse();
    const archEl = document.getElementById('weekly-archive');
    if (!weeks.length) {
        archEl.innerHTML = '<div class="empty-state"><div class="empty-icon">&#128250;</div>No past weeks yet — once Monday rolls around, this week\'s topic moves here.</div>';
        return;
    }
    let archHtml = '';
    for (const wk of weeks) {
        const d = new Date(wk + 'T00:00:00');
        const lbl = d.toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' });
        const entry = byWeek[wk];
        archHtml += '<div class="weekly-archive">';
        archHtml += '<div class="week-label">Week of ' + lbl + '</div>';
        archHtml += '<div class="topic">' + (entry.topic ? escapeHtml(entry.topic.text) : '<span style="color:var(--text-muted);font-weight:500;font-style:italic;">No topic was set</span>') + '</div>';
        archHtml += '<div class="weekly-archive-answers">';
        for (const p of ['douwe', 'ilian']) {
            const a = entry.answers[p];
            const lblP = p === 'douwe' ? 'Douwe' : 'Ilian';
            archHtml += '<div class="weekly-archive-answer ' + p + '">';
            archHtml += '<div class="who-line">' + lblP + '</div>';
            if (a) archHtml += '<div class="text">' + escapeHtml(a.text) + '</div>';
            else archHtml += '<div class="empty-text">No answer.</div>';
            archHtml += '</div>';
        }
        archHtml += '</div></div>';
    }
    archEl.innerHTML = archHtml;
}
function getISOWeek(d) {
    const date = new Date(d);
    date.setHours(0,0,0,0);
    date.setDate(date.getDate() + 3 - (date.getDay() + 6) % 7);
    const week1 = new Date(date.getFullYear(), 0, 4);
    return 1 + Math.round(((date - week1) / 86400000 - 3 + (week1.getDay() + 6) % 7) / 7);
}
function editWeeklyTopic() {
    const items = (state.weeklies || []).filter(w => !w.deleted);
    const thisWeek = getWeekStartISO(new Date());
    const topicItems = items.filter(w => w.kind === 'topic' && w.week_start === thisWeek)
                            .sort((a,b) => (b.updated_at||0) - (a.updated_at||0));
    const current = topicItems[0];
    document.getElementById('weekly-topic-display').style.display = 'none';
    document.getElementById('weekly-topic-editor').style.display = 'flex';
    document.getElementById('weekly-topic-input').value = current ? current.text : '';
    document.getElementById('weekly-topic-input').focus();
}
function cancelWeeklyTopic() { renderWeekly(); }
function saveWeeklyTopic() {
    const txt = document.getElementById('weekly-topic-input').value.trim();
    if (!txt) { renderWeekly(); return; }
    if (!state.weeklies) state.weeklies = [];
    const thisWeek = getWeekStartISO(new Date());
    // Soft-delete any existing topic for this week
    for (const w of state.weeklies) {
        if (!w.deleted && w.kind === 'topic' && w.week_start === thisWeek) {
            w.deleted = true;
            w.updated_at = nowTs();
        }
    }
    state.weeklies.push({
        id: uid(),
        kind: 'topic',
        week_start: thisWeek,
        text: txt,
        person: currentPerson,
        updated_at: nowTs(),
    });
    saveState();
    renderWeekly();
}
function saveWeeklyAnswer(person) {
    if (person !== currentPerson) return;
    const txt = document.getElementById('weekly-ans-' + person).value.trim();
    if (!state.weeklies) state.weeklies = [];
    const thisWeek = getWeekStartISO(new Date());
    // Soft-delete existing answer
    for (const w of state.weeklies) {
        if (!w.deleted && w.kind === 'answer' && w.week_start === thisWeek && w.person === person) {
            w.deleted = true;
            w.updated_at = nowTs();
        }
    }
    if (txt) {
        state.weeklies.push({
            id: uid(),
            kind: 'answer',
            week_start: thisWeek,
            person: person,
            text: txt,
            updated_at: nowTs(),
        });
    }
    saveState();
    renderWeekly();
}

            + '<div class="meta">' + topicTypeLabel(t.type)

function checkAchievements() {
    if (!state.achievements) state.achievements = [];
    const unlockedKeys = new Set(state.achievements.map(a => a.key));
    const newOnes = [];
    for (const a of ACHIEVEMENTS) {
        if (unlockedKeys.has(a.key)) continue;
        try {
            if (a.check(state)) {
                state.achievements.push({ id: uid(), key: a.key, unlocked_at: nowTs(), person: currentPerson });
                newOnes.push(a);
            }
        } catch (e) { /* ignore */ }
    }
    if (newOnes.length) {
        localStorage.setItem(STORE_KEY, JSON.stringify(state));
        for (const a of newOnes) showAchievementToast(a);
    }
}

}

        if (!unlocked[a.key] || a.unlocked_at < unlocked[a.key].unlocked_at) unlocked[a.key] = a;
    }
    const summary = document.getElementById('ach-summary');
    const total = ACHIEVEMENTS.length;
    const got = Object.keys(unlocked).length;
    summary.innerHTML = '<div class="count">' + got + ' / ' + total + '</div>'
        + '<div class="lbl">unlocked &middot; ' + Math.round(got/total*100) + '% complete</div>';
    const grid = document.getElementById('ach-grid');
    grid.innerHTML = ACHIEVEMENTS.map(a => {
        const u = unlocked[a.key];
        const cls = u ? 'unlocked' : 'locked';
        const date = u ? new Date(u.unlocked_at).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) : '';
        return '<div class="ach-card ' + cls + '">'
            + '<div class="ach-icon">' + (u ? a.icon : '🔒') + '</div>'
            + '<div class="ach-name">' + escapeHtml(a.name) + '</div>'
            + '<div class="ach-desc">' + escapeHtml(a.desc) + '</div>'
            + (u ? '<div class="ach-date">Unlocked ' + date + '</div>' : '')
            + '</div>';
    }).join('');
}
