// === KURDISH CONFETTI ===
function kurdishConfetti(opts) {
    opts = opts || {};
    const originX = (opts.x !== undefined) ? opts.x : window.innerWidth / 2;
    const originY = (opts.y !== undefined) ? opts.y : window.innerHeight / 2;
    const scale = opts.scale || 1.0;
    const colors = ['#c1432e', '#6b8e9e', '#c9b8a4', '#e08e3c', '#d4a373', '#7a9b76'];
    const container = document.createElement('div');
    container.className = 'confetti-container';
    document.body.appendChild(container);

    if (!state.counters) state.counters = {};
    state.counters.confetti = (state.counters.confetti || 0) + 1;

    // Symmetric 4-quadrant burst: pick random offsets and mirror across both axes
    const PAIRS = Math.round(14 * scale);
    for (let i = 0; i < PAIRS; i++) {
        const dx = (40 + Math.random() * 180) * scale;
        const dy = (30 + Math.random() * 160) * scale;
        const size = 5 + Math.random() * 7;
        const color = colors[Math.floor(Math.random() * colors.length)];
        for (const sx of [1, -1]) for (const sy of [1, -1]) {
            const piece = document.createElement('div');
            piece.className = 'confetti-piece';
            piece.style.left = (originX - size/2) + 'px';
            piece.style.top  = (originY - size/2) + 'px';
            piece.style.width = size + 'px';
            piece.style.height = size + 'px';
            piece.style.background = color;
            piece.style.setProperty('--dx', (dx * sx) + 'px');
            piece.style.setProperty('--dy', (dy * sy) + 'px');
            piece.style.setProperty('--rot', (Math.random() * 360 - 180) + 'deg');
            piece.style.animation = 'confettiFly ' + (0.9 + Math.random() * 0.7) + 's ease-out forwards';
            container.appendChild(piece);
        }
    }
    // A few "central diamond" pieces (no mirror) for extra texture
    for (let i = 0; i < 6; i++) {
        const ang = (Math.PI * 2 / 6) * i;
        const d = 60 * scale;
        const size = 5;
        const color = colors[i % colors.length];
        const piece = document.createElement('div');
        piece.className = 'confetti-piece';
        piece.style.left = (originX - size/2) + 'px';
        piece.style.top  = (originY - size/2) + 'px';
        piece.style.width = size + 'px';
        piece.style.height = size + 'px';
        piece.style.background = color;
        piece.style.setProperty('--dx', (Math.cos(ang) * d) + 'px');
        piece.style.setProperty('--dy', (Math.sin(ang) * d) + 'px');
        piece.style.animation = 'confettiFly 0.7s ease-out forwards';
        container.appendChild(piece);
    }
    setTimeout(() => container.remove(), 2200);
}

// === TIME OF DAY ===
function applyTimeOfDay() {
    const h = new Date().getHours();
    let phase;
    if (h < 6 || h >= 21) phase = 'night';
    else if (h < 9) phase = 'dawn';
    else if (h < 18) phase = 'day';
    else phase = 'dusk';
    document.body.dataset.tod = phase;
    if (!state.counters) state.counters = {};
    if (!state.counters.tod_seen) state.counters.tod_seen = {};
    state.counters.tod_seen[phase] = true;
}

// === COUNTDOWN (multi-target) ===
function renderCountdown() {
    const els = document.querySelectorAll('.countdown-widget');
    if (!els.length) return;
    const now = new Date();
    const diffMs = COUNTDOWN_TARGET.getTime() - now.getTime();
    const days = Math.max(0, Math.ceil(diffMs / 86400000));
    if (!state.counters) state.counters = {};
    state.counters.countdown_views = (state.counters.countdown_views || 0) + 1;
    const html = (days <= 0)
        ? '<div class="countdown-content">'
            + '<div class="countdown-days">&#127881;</div>'
            + '<div><div style="font-weight:700;color:var(--primary-dark);">The day is here!</div>'
            + '<div class="countdown-label">September 1st 2026</div></div></div>'
        : '<div class="countdown-content">'
            + '<div class="countdown-days">' + days + '</div>'
            + '<div><div style="font-weight:700;color:var(--primary-dark);">days to go</div>'
            + '<div class="countdown-label">until we meet again</div></div></div>'
            + '<div class="countdown-target">&#128197;&nbsp;September 1, 2026</div>';
    els.forEach(el => { el.innerHTML = html; });
}

// === INLINE EDIT (task title) ===
function inlineEditTask(id, span) {
    const t = (state.tasks || []).find(x => x.id === id);
    if (!t) return;
    const original = t.title;
    const input = document.createElement('input');
    input.type = 'text';
    input.value = original;
    input.className = 'inline-edit-input';
    span.replaceWith(input);
    input.focus();
    input.select();
    let committed = false;
    const commit = (save) => {
        if (committed) return;
        committed = true;
        const v = input.value.trim();
        if (save && v && v !== original) {
            t.title = v;
            t.updated_at = nowTs();
            if (!state.counters) state.counters = {};
            state.counters.inline_edits = (state.counters.inline_edits || 0) + 1;
            saveState();
        }
        renderCalendar();
    };
    input.addEventListener('blur', () => commit(true));
    input.addEventListener('keydown', e => {
        if (e.key === 'Enter') { e.preventDefault(); commit(true); }
        if (e.key === 'Escape') commit(false);
    });
}

// === DRAG-DROP TASKS ===
function taskDragStart(e, id) {
    e.dataTransfer.setData('text/plain', id);
    e.dataTransfer.effectAllowed = 'move';
    e.currentTarget.classList.add('dragging');
}
function taskDragEnd(e) {
    e.currentTarget.classList.remove('dragging');
}
function taskDragOver(e) {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    e.currentTarget.classList.add('drag-target');
}
function taskDragLeave(e) {
    if (e.currentTarget.contains(e.relatedTarget)) return;
    e.currentTarget.classList.remove('drag-target');
}
function taskDrop(e, dateIso) {
    e.preventDefault();
    e.currentTarget.classList.remove('drag-target');
    const id = e.dataTransfer.getData('text/plain');
    const t = (state.tasks || []).find(x => x.id === id);
    if (t && t.date !== dateIso) {
        t.date = dateIso;
        t.updated_at = nowTs();
        if (!state.counters) state.counters = {};
        state.counters.drag_drops = (state.counters.drag_drops || 0) + 1;
        saveState();
        renderCalendar();
    }
}

// === TODAY PAGE ===
function relativeTime(ts) {
    const diff = Date.now() - ts;
    if (diff < 60000) return 'just now';
    if (diff < 3600000) return Math.floor(diff/60000) + 'm ago';
    if (diff < 86400000) return Math.floor(diff/3600000) + 'h ago';
    return Math.floor(diff/86400000) + 'd ago';
}

    renderCountdown();

    // Tasks for today
    const todayTasks = (state.tasks || [])
        .filter(t => !t.deleted && t.date === todayIso)
        .sort((a,b) => (a.time || '').localeCompare(b.time || ''));
    const tasksEl = document.getElementById('today-tasks-list');
    if (!todayTasks.length) {
        tasksEl.innerHTML = '<div class="empty-state" style="padding:20px;"><div class="empty-icon" style="font-size:2em;">\u{1F389}</div>Nothing scheduled for today.</div>';

            + '<input type="checkbox" ' + (t.is_completed ? 'checked' : '') + ' onchange="toggleTaskFromToday(\'' + t.id + '\', this)">'
            + (t.time ? '<span style="font-size:0.78em;color:var(--text-muted);font-weight:600;">' + t.time + '</span>' : '')
            + '<span class="ttitle">' + escapeHtml(t.title) + '</span>'
            + personTag(t.person)
            + '</div>'
        ).join('');
    }

    // Today's songs (compact)
    const songsEl = document.getElementById('today-songs-strip');
    songsEl.innerHTML = ['douwe', 'ilian'].map(p => {

                : '<a href="' + escapeAttr(song.url) + '" target="_blank" rel="noopener" style="font-size:0.85em;">Open in Spotify &rarr;</a>';

                inner += '<div class="backstory-mini">' + escapeHtml(song.backstory) + '</div>';
            }

            inner = '<div class="empty">No pick yet. <a onclick="showPage(\'songs\')" style="color:var(--primary-dark);cursor:pointer;text-decoration:underline;">Add one &rarr;</a></div>';
        }
        return '<div class="today-mini-song ' + p + '">'
            + '<div class="who-line">' + label + '</div>'
            + inner + '</div>';
    }).join('');

    // This week's topic mini
    const thisWeek = getWeekStartISO(new Date());
    const wkItems = (state.weeklies || []).filter(w => !w.deleted);
    const wkTopic = wkItems.filter(w => w.kind === 'topic' && w.week_start === thisWeek)
                           .sort((a,b)=>(b.updated_at||0)-(a.updated_at||0))[0];
    const douweAns = wkItems.find(w => w.kind === 'answer' && w.week_start === thisWeek && w.person === 'douwe');
    const ilianAns = wkItems.find(w => w.kind === 'answer' && w.week_start === thisWeek && w.person === 'ilian');
    const wkEl = document.getElementById('today-weekly-mini');
    if (wkTopic) {
        wkEl.innerHTML = '<div class="topic">' + escapeHtml(wkTopic.text) + '</div>'
            + '<div class="status">'
            + '<span class="pill ' + (douweAns ? 'done' : '') + '">' + (douweAns ? '✓ ' : '· ') + 'Douwe</span>'
            + '<span class="pill ' + (ilianAns ? 'done' : '') + '">' + (ilianAns ? '✓ ' : '· ') + 'Ilian</span>'
            + '</div>';
    } else {
        wkEl.innerHTML = '<div class="status"><em style="color:var(--text-muted);">No topic for this week yet.</em></div>';
    }

    // Recently added (top 8 across all collections)
    const recent = [];
    for (const kind of Object.keys(SECTION_META)) {
        for (const it of (state[kind] || [])) {
            if (it.deleted) continue;
            recent.push({ kind, item: it });
        }
    }
    recent.sort((a,b) => (b.item.updated_at || 0) - (a.item.updated_at || 0));
    const recentEl = document.getElementById('today-recent');
    if (!recent.length) {
        recentEl.innerHTML = '<div style="color:var(--text-muted);font-size:0.85em;font-style:italic;">Nothing added yet.</div>';
    } else {
        recentEl.innerHTML = recent.slice(0, 8).map(r => {
            const meta = SECTION_META[r.kind];
            const title = meta.getTitle(r.item);
            return '<div class="today-recent-item">'
                + '<div class="icon">' + meta.icon + '</div>'
                + '<div class="title">' + escapeHtml(title || '') + '</div>'
                + '<div class="meta">' + (r.item.person ? (r.item.person === 'douwe' ? 'Douwe' : 'Ilian') : '') + ' &middot; ' + relativeTime(r.item.updated_at || 0) + '</div>'
                + '</div>';
        }).join('');
    }

    // Fresh topics (last few non-discussed)
    const fresh = (state.topics || [])
        .filter(t => !t.deleted && !t.discussed_at)
        .sort((a,b) => (b.updated_at||0) - (a.updated_at||0))
        .slice(0, 5);
    const ftEl = document.getElementById('today-fresh-topics');
    if (!fresh.length) {
        ftEl.innerHTML = '<div style="color:var(--text-muted);font-size:0.85em;font-style:italic;">Pool is empty.</div>';
    } else {
        ftEl.innerHTML = fresh.map(t =>
            '<div class="ft-item ' + (t.person || '') + '">'
            + escapeHtml(t.text)
            + '<div class="ft-meta">' + topicTypeLabel(t.type) + ' &middot; ' + relativeTime(t.updated_at || 0) + '</div>'
            + '</div>'
        ).join('');
    }
}

function toggleTaskFromToday(id, checkbox) {
    const t = (state.tasks || []).find(x => x.id === id);
    if (!t) return;
    t.is_completed = !t.is_completed;
    t.updated_at = nowTs();
    if (t.is_completed && checkbox) {
        const rect = checkbox.getBoundingClientRect();
        kurdishConfetti({ x: rect.left + rect.width/2, y: rect.top + rect.height/2, scale: 0.6 });
    }
    saveState();
    renderToday();
}

function renderStats() {
    const el = document.getElementById('stats-row');
    if (!el) return;
    const tasks = (state.tasks || []).filter(t => !t.deleted);
    const tasksDone = tasks.filter(t => t.is_completed).length;
    const books = (state.books || []).filter(b => !b.deleted);
    const pagesRead = books.reduce((a,b) => a + (b.pages_read || 0), 0);
    const booksDone = books.filter(b => b.is_finished).length;
    const movies = (state.movies || []).filter(m => !m.deleted);
    const moviesWatched = movies.filter(m => m.watched).length;
    const series = (state.series || []).filter(s => !s.deleted);
    const seriesDone = series.filter(s => s.total_episodes > 0 && s.episodes_watched >= s.total_episodes).length;
    const songsTotal = (state.songs || []).filter(s => !s.deleted).length;
    const songDays = new Set((state.songs || []).filter(s => !s.deleted).map(s => s.date)).size;
    const backstories = (state.songs || []).filter(s => !s.deleted && s.backstory && s.backstory.trim()).length;
    const topicsTotal = (state.topics || []).filter(t => !t.deleted).length;
    const topicsDiscussed = (state.topics || []).filter(t => t.discussed_at).length;
    const spins = (state.counters && state.counters.spins) || 0;
    const weekliesAns = (state.weeklies || []).filter(w => !w.deleted && w.kind === 'answer').length;
    const drags = (state.counters && state.counters.drag_drops) || 0;
    const tiles = [
        ['\u{1F4C5}', tasksDone + '/' + tasks.length, 'tasks done'],
        ['\u{1F4D6}', pagesRead.toLocaleString(), 'pages read'],
        ['\u{1F4DA}', booksDone, 'books finished'],
        ['\u{1F3AC}', moviesWatched, 'movies watched'],
        ['\u{1F4FA}', seriesDone, 'series finished'],
        ['\u{1F3B5}', songsTotal, 'songs picked'],
        ['\u{1F4C6}', songDays, 'music days'],
        ['\u{1F4DD}', backstories, 'backstories'],
        ['\u{1F4AC}', topicsTotal, 'topics in pool'],
        ['\u{1F5E3}', topicsDiscussed, 'discussed'],
        ['\u{1F3B0}', spins, 'spins'],
        ['\u{1F4C5}', weekliesAns, 'weekly answers'],
    ];
    el.innerHTML = tiles.map(t =>
        '<div class="stat-cell"><div class="stat-icon">' + t[0] + '</div>'
        + '<div class="stat-value">' + t[1] + '</div>'
        + '<div class="stat-label">' + t[2] + '</div></div>'
    ).join('');
}
