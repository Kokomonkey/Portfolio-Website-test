// === THEMES ===
const THEMES = ['cozy', 'forest', 'ocean', 'midnight'];

function setTheme(name) {
    if (!THEMES.includes(name)) name = 'cozy';
    for (const t of THEMES) document.body.classList.remove('theme-' + t);
    document.body.classList.add('theme-' + name);
    localStorage.setItem('kloef_theme', name);
    document.querySelectorAll('.theme-opt').forEach(b => {
        b.classList.toggle('selected', b.dataset.theme === name);
    });
    if (!state.counters) state.counters = {};
    state.counters.theme_changes = (state.counters.theme_changes || 0) + 1;
    if (name === 'midnight') state.counters.midnight_used = true;
    applyTimeOfDay();
    saveState();
}

function initTheme() {
    const saved = localStorage.getItem('kloef_theme') || 'cozy';
    for (const t of THEMES) document.body.classList.remove('theme-' + t);
    document.body.classList.add('theme-' + saved);
    document.querySelectorAll('.theme-opt').forEach(b => {
        b.classList.toggle('selected', b.dataset.theme === saved);
        b.addEventListener('click', () => setTheme(b.dataset.theme));
    });
}

// === UNDO BAR ===
let undoTimer = null;

function showUndoBar(kind, id, label) {
    const existing = document.getElementById('undo-bar');
    if (existing) existing.remove();
    if (undoTimer) clearTimeout(undoTimer);
    const bar = document.createElement('div');
    bar.id = 'undo-bar';
    bar.className = 'undo-bar';
    bar.innerHTML = '<span class="undo-msg">' + escapeHtml(label) + '</span>'
        + '<button class="undo-btn" onclick="undoDelete(\'' + kind + '\',\'' + id + '\')">Undo</button>';
    document.body.appendChild(bar);
    undoTimer = setTimeout(() => {
        bar.classList.add('leaving');
        setTimeout(() => bar.remove(), 300);
    }, 5000);
}

function undoDelete(kind, id) {
    const it = (state[kind] || []).find(x => x.id === id);
    if (it) {
        it.deleted = false;
        it.updated_at = nowTs();
        if (!state.counters) state.counters = {};
        state.counters.undos = (state.counters.undos || 0) + 1;
        saveState();
        rerenderActive();
    }
    const bar = document.getElementById('undo-bar');
    if (bar) bar.remove();
    if (undoTimer) clearTimeout(undoTimer);
}
