// === QUOTES ===
const QUOTE_COLORS = ['yellow', 'pink', 'blue', 'green', 'orange', 'purple', 'cream'];

function hashCode(s) {
    let h = 0;
    for (let i = 0; i < s.length; i++) h = ((h << 5) - h) + s.charCodeAt(i) | 0;
    return Math.abs(h);
}

function addQuote(e) {
    e.preventDefault();
    const f = e.target;
    if (!state.quotes) state.quotes = [];
    state.quotes.push({
        id: uid(),
        text: f.text.value.trim(),
        author: f.author.value.trim(),
        source: f.source.value.trim(),
        person: currentPerson,
        updated_at: nowTs(),
    });
    saveState();
    f.reset();
    renderQuotes();
}

function renderQuotes() {
    if (!state.quotes) state.quotes = [];
    const items = state.quotes.filter(q => !q.deleted)
                              .sort((a,b) => (b.updated_at||0) - (a.updated_at||0));
    const board = document.getElementById('quotes-board');
    if (!items.length) {
        board.innerHTML = '<div style="grid-column: 1/-1; text-align:center; color:#fff5db; padding: 60px 20px; font-family: Georgia, serif;">'
            + '<div style="font-size:2.5em;">&#128209;</div>'
            + 'The board is empty. Pin the first one above.</div>';
        return;
    }
    let html = '';
    items.forEach((q, i) => {
        const h = hashCode(q.id);
        const color = QUOTE_COLORS[h % QUOTE_COLORS.length];
        const rot = ((h % 7) - 3) + (Math.random() * 0.4 - 0.2);
        const label = q.person === 'douwe' ? 'Douwe' : (q.person === 'ilian' ? 'Ilian' : '');
        html += '<div class="sticky-note ' + color + '" style="--rot:' + rot.toFixed(2) + 'deg; --delay:' + (i * 0.04) + 's;">';
        html += '<div class="qtext">' + escapeHtml(q.text) + '</div>';
        if (q.author) html += '<div class="qauthor">&mdash; ' + escapeHtml(q.author) + '</div>';
        if (q.source) html += '<div class="qsource">' + escapeHtml(q.source) + '</div>';
        html += '<div class="qfoot">';
        html += '<span class="qpersonchip ' + (q.person || '') + '">' + escapeHtml(label) + '</span>';
        html += '<button class="qdel" onclick="deleteItem(\'quotes\',\'' + q.id + '\')" title="Remove">&times;</button>';
        html += '</div></div>';
    });
    board.innerHTML = html;
}

function getQuoteOfDay() {
    const quotes = (state.quotes || []).filter(q => !q.deleted);
    if (!quotes.length) return null;
    const todayIso = fmtISODate(new Date());
    const h = hashCode(todayIso);
    return quotes[h % quotes.length];
}

function renderQuoteOfDay() {
    const card = document.getElementById('today-quote-card');
    if (!card) return;
    const q = getQuoteOfDay();
    if (!q) {
        card.style.display = 'none';
        return;
    }
    card.style.display = 'block';
    card.className = 'today-quote-card';
    let html = '<div class="qlabel">&#127775; Quote of the day</div>';
    html += '<div class="qbody">' + escapeHtml(q.text) + '</div>';
    html += '<div class="qmeta">&mdash; ' + escapeHtml(q.author || 'Anonymous');
    if (q.source) html += '<span class="src">from ' + escapeHtml(q.source) + '</span>';
    html += '</div>';
    card.innerHTML = html;
    // Mark that this quote's author has been featured (for achievement)
    if (!state.counters) state.counters = {};
    if (!state.counters.qod_seen) state.counters.qod_seen = {};
    state.counters.qod_seen[q.id] = true;
}
